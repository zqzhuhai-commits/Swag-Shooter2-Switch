cmake_minimum_required(VERSION 3.22)
project(swag2_armv7_probe LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(DYNARMIC_FRONTENDS A32 CACHE STRING "" FORCE)
set(DYNARMIC_TESTS OFF CACHE BOOL "" FORCE)
set(DYNARMIC_WARNINGS_AS_ERRORS OFF CACHE BOOL "" FORCE)
set(DYNARMIC_USE_BUNDLED_EXTERNALS ON CACHE BOOL "" FORCE)
set(DYNARMIC_USE_PRECOMPILED_HEADERS OFF CACHE BOOL "" FORCE)
if(CMAKE_SYSTEM_NAME STREQUAL "Generic")
  set(DYNARMIC_ENABLE_NO_EXECUTE_SUPPORT ON CACHE BOOL "" FORCE)
endif()
if(NOT DEFINED DYNARMIC_SOURCE)
  message(FATAL_ERROR "Set DYNARMIC_SOURCE to the pinned, patched Dynarmic checkout")
endif()
add_subdirectory(${DYNARMIC_SOURCE} dynarmic)
add_executable(swag2-armv7-test src/main.cpp src/probe.cpp)
target_link_libraries(swag2-armv7-test PRIVATE dynarmic)
target_compile_options(swag2-armv7-test PRIVATE -Wall -Wextra -Werror)
if(CMAKE_SYSTEM_NAME STREQUAL "Generic")
  target_include_directories(swag2-armv7-test SYSTEM PRIVATE /opt/devkitpro/libnx/include)
  # libnx C headers intentionally use partially initialized aggregate literals.
  target_compile_options(swag2-armv7-test PRIVATE -Wno-missing-field-initializers)
  target_link_directories(swag2-armv7-test PRIVATE /opt/devkitpro/libnx/lib)
  target_link_options(swag2-armv7-test PRIVATE -specs=/opt/devkitpro/libnx/switch.specs)
  target_link_libraries(swag2-armv7-test PRIVATE nx)
  add_custom_command(TARGET swag2-armv7-test POST_BUILD
    COMMAND /opt/devkitpro/tools/bin/nacptool --create "SWAG2 ARMv7 execution test" "Xiao Fan" "0.0.1" ${CMAKE_CURRENT_BINARY_DIR}/swag2.nacp
    COMMAND /opt/devkitpro/tools/bin/elf2nro $<TARGET_FILE:swag2-armv7-test> ${CMAKE_CURRENT_BINARY_DIR}/swag2-armv7-test.nro --nacp=${CMAKE_CURRENT_BINARY_DIR}/swag2.nacp)
endif()
